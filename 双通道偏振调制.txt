        img_path = self.img_paths[idx]

        label_one_hot[label] = 1.0

       # img = img.astype(np.float32)
        img = img.astype(np.float32) / 255.0   # 归一化到 [0,1]

        img = np.expand_dims(img, axis=0)
        img_tensor = torch.from_numpy(img)
        label_tensor = torch.from_numpy(label_one_hot)
        return img_tensor, label_tensor

    def __len__(self):
        return len(self.img_names)

def pixel2amplitude(X):
    X = torch.sqrt(X)
    return X

# 自定义预处理层
class pre_process(torch.nn.Module):
    """自定义预处理层。"""
    def __init__(self):
        super(pre_process, self).__init__()

    def forward(self, X):
        # X = X / 255
        # X = pixel2amplitude(X)
        
        X = X * (2 * np.pi) #相位取值为0~2pi
        X = torch.exp(1j * X)
        X = torch.nn.functional.pad(X, (window_expand, window_expand, window_expand, window_expand), "constant", 0)
        return X

# 衍射传输层
class DiffractionTransmission(torch.nn.Module):
    """衍射传输层，用于计算光场的衍射传输过程."""
    def __init__(self, D):
        super(DiffractionTransmission, self).__init__()
        self.D = D
        FreqPoint1 = torch.linspace(-1/(2 * neuron), 1/(2 * neuron), sampling_point + 2 * window_expand, device=device)
        FreqPoint2 = torch.linspace(-1/(2 * neuron), 1/(2 * neuron), sampling_point + 2 * window_expand, device=device)
        self.fu, self.fv = torch.meshgrid(FreqPoint1, FreqPoint2, indexing='ij')

    def sqrt(self, X):
        """自定义开方函数"""
        y = torch.where(X < 0, torch.zeros_like(X), torch.sqrt(X))
        return y

    def forward(self, X):
        dif = (2 * np.pi / wave_length) * self.sqrt(2.1316 - torch.square(wave_length * self.fu) - torch.square(wave_length * self.fv))
        angular_spectrum = torch.fft.fftshift(torch.fft.fft2(X)) * torch.where(torch.square(wave_length * self.fu) + torch.square(wave_length * self.fv) < 2.1316, 1, 0) * torch.exp(1j * dif * self.D)
        X = torch.fft.ifft2(torch.fft.ifftshift(angular_spectrum))
        return X
class DiffractionTransmission1(torch.nn.Module):
    """衍射传输层，用于计算光场的衍射传输过程."""
    def __init__(self, D):
        super(DiffractionTransmission1, self).__init__()
        self.D = D
        FreqPoint1 = torch.linspace(-1/(2 * neuron), 1/(2 * neuron), sampling_point + 2 * window_expand, device=device )
        FreqPoint2 = torch.linspace(-1/(2 * neuron), 1/(2 * neuron), sampling_point + 2 * window_expand, device=device )
        self.fu, self.fv = torch.meshgrid(FreqPoint1, FreqPoint2, indexing='ij')

    def sqrt(self, X):
        """自定义开方函数"""
        y = torch.where(X < 0, torch.zeros_like(X), torch.sqrt(X))
        return y

    def forward(self, X):
        dif = (2 * np.pi / wave_length) * self.sqrt(1 - torch.square(wave_length * self.fu) - torch.square(wave_length * self.fv))
        angular_spectrum = torch.fft.fftshift(torch.fft.fft2(X)) * torch.where(torch.square(wave_length * self.fu) + torch.square(wave_length * self.fv) < 1, 1, 0) * torch.exp(1j * dif * self.D)
        X = torch.fft.ifft2(torch.fft.ifftshift(angular_spectrum))
        return X
def param2phase(param):
    """将参数映射到相位值"""
    phase = torch.sigmoid(param) * (2 * np.pi)
    return phase

class modulate_a(torch.nn.Module):
    """光场相位调制层"""
    def __init__(self):
        super(modulate_a, self).__init__()
        self.M = torch.nn.Parameter(torch.zeros(sampling_point, sampling_point))  # 创建参数张量
# =============================================================================
#         with torch.no_grad():
#             # 生成固定噪声矩阵（标准差5，最大约±15）
#             self.gamma_noise = torch.randn(
#                 sampling_point, sampling_point,
#                 device=device
#             ) * 5
# =============================================================================
    def forward(self, X, polarization):
        """
        X: 入射光场
        polarization: 'RCP' 或 'LCP'，表示左右旋偏振光
        """
# =============================================================================
#         gamma_noisy = gamma + self.gamma_noise
#         # 将gamma矩阵扩展到与填充后相同的尺寸
#         padded_gamma = torch.nn.functional.pad(
#             gamma_noisy,
#             (window_expand, window_expand, window_expand, window_expand),
#             "constant",
#             0
#         )
# =============================================================================
        g = torch.tensor(gamma / 2.0, dtype=torch.float32)
        if polarization == 'LCP':
            term1 = X * torch.cos(g)
            term2 = -1j * X * torch.sin(g) * torch.exp(-1j * torch.nn.functional.pad(param2phase(self.M), (window_expand, window_expand, window_expand, window_expand), "constant", 0))
        elif polarization == 'RCP':
            term1 = -1j * X * torch.sin(g) * torch.exp(1j * torch.nn.functional.pad(param2phase(self.M), (window_expand, window_expand, window_expand, window_expand), "constant", 0))
            term2 = X * torch.cos(g)

        return term1, term2  # 分别返回左右旋光的电场
class modulate_b(torch.nn.Module):
    """光场相位调制层"""
    def __init__(self):
        super(modulate_b, self).__init__()
        self.M = torch.nn.Parameter(torch.zeros(sampling_point, sampling_point))  # 创建参数张量
# =============================================================================
#         with torch.no_grad():
#             # 生成固定噪声矩阵（标准差5，最大约±15）
#             self.gamma_noise = torch.randn(
#                 sampling_point, sampling_point,
#                 device=device
#             ) * 5
# =============================================================================
    def forward(self, X_LCP, X_RCP):
        """
        X_LCP: 左旋圆偏振光
        X_RCP: 右旋圆偏振光
        """
# =============================================================================
#         gamma_noisy = gamma + self.gamma_noise
# 
#         # 将gamma矩阵扩展到与填充后相同的尺寸
#         padded_gamma = torch.nn.functional.pad(
#             gamma_noisy,
#             (window_expand, window_expand, window_expand, window_expand),
#             "constant",
#             0
#         )
# =============================================================================
        g = torch.tensor(gamma / 2.0, dtype=torch.float32)
        # 对左旋圆偏振光（LCP）进行调制
        term1_LCP = X_LCP * torch.cos(g)
        term2_LCP = -1j * X_LCP * torch.sin(g) * torch.exp(-1j * torch.nn.functional.pad(param2phase(self.M), (window_expand, window_expand, window_expand, window_expand), "constant", 0))

        # 对右旋圆偏振光（RCP）进行调制
        term1_RCP = -1j * X_RCP * torch.sin(g) * torch.exp(1j * torch.nn.functional.pad(param2phase(self.M), (window_expand, window_expand, window_expand, window_expand), "constant", 0))
        term2_RCP = X_RCP * torch.cos(g)

        # 将所有左旋光项叠加
        term_LCP = term1_LCP + term1_RCP
        term_RCP = term2_LCP + term2_RCP

        return term_LCP, term_RCP

class after_process(torch.nn.Module):
    """用于获取最终输出光场的后处理。"""
    def __init__(self):
        super(after_process, self).__init__()

    def forward(self, X):
        X = X.abs().square()
        return X

class Detector(nn.Module):
    def __init__(self, num_classes=10, region_size=10, radius=None, start_angle_deg=0.0):
        super(Detector, self).__init__()
        self.num_classes = num_classes
        self.region_size = region_size
        self.radius = radius
        self.start_angle = np.deg2rad(start_angle_deg)

    def _gen_coords(self, H, W, device):
        cy, cx = H // 2, W // 2
        r = self.radius
        if r is None:
            r = min(H, W) // 2 - self.region_size - 5
            r = max(r, self.region_size)

        coords = []
        for k in range(self.num_classes):
            angle = self.start_angle + 2 * np.pi * k / self.num_classes
            y_c = cy + int(round(r * np.sin(angle)))
            x_c = cx + int(round(r * np.cos(angle)))
            y0 = int(y_c - self.region_size // 2)
            x0 = int(x_c - self.region_size // 2)
            y0 = max(0, min(y0, H - self.region_size))
            x0 = max(0, min(x0, W - self.region_size))
            coords.append((y0, x0, self.region_size, self.region_size))
        return coords

    def get_region_masks(self, H, W, device):
        """
        返回:
        masks: [num_classes, 1, H, W]
        """
        coords = self._gen_coords(H, W, device)
        masks = []
        for (y0, x0, h, w) in coords:
            mask = torch.zeros(1, H, W, device=device)
            mask[:, y0:y0+h, x0:x0+w] = 1.0
            masks.append(mask)
        masks = torch.stack(masks, dim=0)   # [C,1,H,W]
        return masks

    def get_all_regions_mask(self, H, W, device):
        region_masks = self.get_region_masks(H, W, device)   # [C,1,H,W]
        all_mask = torch.clamp(region_masks.sum(dim=0, keepdim=False), 0, 1)  # [1,H,W]
        return all_mask

    def forward(self, x):
        assert x.dim() == 4 and x.size(1) == 1, f"Detector expects [B,1,H,W], got {list(x.shape)}"
        B, _, H, W = x.shape
        coords = self._gen_coords(H, W, x.device)

        regions = []
        for (y, x0, h, w) in coords:
            region = x[:, :, y:y+h, x0:x0+w]
            avg_intensity = region.mean(dim=(2, 3)).squeeze(1)
            regions.append(avg_intensity.unsqueeze(1))

        out = torch.cat(regions, dim=1)
        return out

class D2NN(torch.nn.Module):
    def __init__(self, distance):
        super(D2NN, self).__init__()
        self.pre_process = pre_process()
        self.transin1 = DiffractionTransmission1(0.05)#空气传播3cm
        self.transin2 = DiffractionTransmission(0.001)#玻璃传播500微米
        self.modulate1 = modulate_a()
        self.transin3 = DiffractionTransmission(0.002)#层间距1毫米
        self.modulate2 = modulate_b()
        self.transin4 = DiffractionTransmission(0.002)#玻璃传播1毫米
        self.modulate3 = modulate_b()
        self.transin5 = DiffractionTransmission(0.001)#介质传播500微米
        self.transin6 = DiffractionTransmission1(0.05)#空气传播3cm
        self.after_process = after_process()
        self.detector = Detector(num_classes=10, region_size=10, radius=80, start_angle_deg=90.0)

    def forward(self, X):
        X = self.pre_process(X)
        X = self.transin1(X)

        # 模拟右旋偏振光 (RCP) 和左旋偏振光 (LCP) 独立传播
        X = self.transin2(X)
        X_LCP, X_RCP = self.modulate1(X, 'LCP')
        X_LCP = self.transin3(X_LCP)
        X_RCP = self.transin3(X_RCP)

        X_LCP, X_RCP = self.modulate2(X_LCP, X_RCP)
        X_LCP = self.transin4(X_LCP)
        X_RCP = self.transin4(X_RCP)
        
        X_LCP, X_RCP = self.modulate3(X_LCP, X_RCP)
        
        X_LCP = self.transin5(X_LCP)
        X_RCP = self.transin5(X_RCP)
        
        X_LCP = self.transin6(X_LCP)
        X_RCP = self.transin6(X_RCP)

        intensity_a = self.after_process(X_LCP)
        intensity_b = self.after_process(X_RCP)

        
        detection_a = self.detector(intensity_a)
        detection_b = self.detector(intensity_b)
        
        diff = (detection_a - detection_b) / (detection_a + detection_b + 1e-8)
        # diff = diff / 0.1
        

        
        return diff, intensity_a, intensity_b, detection_a, detection_b  # 返回合成的强度和图像

def splitting_ratio_loss(intensity_a, intensity_b, detector, label_onehot):
    """
    鼓励正负两个通道都把能量集中到目标探测子区域内，
    同时压低目标子区域外的背景能量。

    intensity_a, intensity_b: [B,1,H,W]
    label_onehot: [B,C]
    """
    B, _, H, W = intensity_a.shape
    device = intensity_a.device
    eps = 1e-8

    # [C,1,H,W]
    region_masks = detector.get_region_masks(H, W, device)
    # [1,H,W]
    all_regions_mask = detector.get_all_regions_mask(H, W, device)

    # [B,C,1,H,W]
    region_masks_b = region_masks.unsqueeze(0).repeat(B, 1, 1, 1, 1)
    # [B,1,H,W] 目标类别对应子区域
    target_mask = torch.sum(label_onehot[:, :, None, None, None] * region_masks_b, dim=1)

    # [B,1,H,W] 非目标区域 = 全图 - 目标区
    outside_target_mask = 1.0 - target_mask

    # ---------------------------
    # 1. 正通道：目标区内能量占比越大越好
    # ---------------------------
    eta_a = (intensity_a * target_mask).sum(dim=(1, 2, 3)) / (intensity_a.sum(dim=(1, 2, 3)) + eps)

    # ---------------------------
    # 2. 负通道：目标区内能量占比也越大越好
    # ---------------------------
    eta_b = (intensity_b * target_mask).sum(dim=(1, 2, 3)) / (intensity_b.sum(dim=(1, 2, 3)) + eps)

    # ---------------------------
    # 3. 子区域外背景占比越小越好
    # ---------------------------
    bg_a = (intensity_a * outside_target_mask).sum(dim=(1, 2, 3)) / (intensity_a.sum(dim=(1, 2, 3)) + eps)
    bg_b = (intensity_b * outside_target_mask).sum(dim=(1, 2, 3)) / (intensity_b.sum(dim=(1, 2, 3)) + eps)

    # ---------------------------
    # 4. 所有探测子区域外的全局背景也压低
    # ---------------------------
    bg_mask = 1.0 - all_regions_mask.unsqueeze(0)   # [B,1,H,W]
    eta_bg = ((intensity_a + intensity_b) * bg_mask).sum(dim=(1, 2, 3)) / (
        (intensity_a + intensity_b).sum(dim=(1, 2, 3)) + eps
    )

    # ---------------------------
    # 5. 总损失
    # 鼓励 a、b 都进入目标区域，同时压低目标区外背景和全局背景
    # ---------------------------
    loss = (1.0 - eta_a) + (1.0 - eta_b) + 0.5 * (bg_a + bg_b) + 0.5 * eta_bg

    return loss.mean()






def cross_entropy_one_hot(input, target):
    log_probs = torch.nn.functional.log_softmax(input, dim = 1)
    loss = -torch.sum(target * log_probs) / input.size(0)
    return loss

def detector_confinement_loss(intensity_a, intensity_b, detector):
    """
    约束正负两个探测面的总能量尽可能集中在所有探测子区域内部，
    并且要求子区域内总能量显著大于子区域外背景能量。

    intensity_a, intensity_b: [B,1,H,W]
    """
    B, _, H, W = intensity_a.shape
    device = intensity_a.device
    eps = 1e-8

    # 所有探测子区域总mask: [1,H,W]
    all_regions_mask = detector.get_all_regions_mask(H, W, device)   # [1,H,W]
    all_regions_mask = all_regions_mask.unsqueeze(0)                  # [B,1,H,W]
    bg_mask = 1.0 - all_regions_mask                                 # [B,1,H,W]

    # 正通道：区域内 / 区域外
    region_energy_a = (intensity_a * all_regions_mask).sum(dim=(1, 2, 3))
    bg_energy_a     = (intensity_a * bg_mask).sum(dim=(1, 2, 3))

    # 负通道：区域内 / 区域外
    region_energy_b = (intensity_b * all_regions_mask).sum(dim=(1, 2, 3))
    bg_energy_b     = (intensity_b * bg_mask).sum(dim=(1, 2, 3))

    # 比值型损失：希望 region >> background
    loss_a = -torch.log(region_energy_a / (bg_energy_a + eps) + eps)
    loss_b = -torch.log(region_energy_b / (bg_energy_b + eps) + eps)

    # 再补一个背景占比项，使背景进一步压低
    total_energy_a = intensity_a.sum(dim=(1, 2, 3))
    total_energy_b = intensity_b.sum(dim=(1, 2, 3))

    bg_ratio_a = bg_energy_a / (total_energy_a + eps)
    bg_ratio_b = bg_energy_b / (total_energy_b + eps)

    loss_bg = bg_ratio_a + bg_ratio_b

    loss = loss_a + loss_b + 0.5 * loss_bg
    return loss.mean()

      all_preds = []

                  all_preds.extend(preds)
                  
                  loss = cross_entropy_one_hot(pred, label)

            valid_loss = epoch_loss / size1

      # 计算准确率
      accuracy = accuracy_score(all_labels, all_preds)

      return accuracy, valid_loss

# %% 显示并保存相位调制层参数
def show_phase_modulators(model, out_dir=r"D:/gongwei/modulate"):
    """
    提取模型中的 modulate1/2/3 层参数 M
    - 保存为 .txt 和 .bmp
    - 显示图像
    """
    os.makedirs(out_dir, exist_ok=True)

    modulators = [model.modulate1.M, model.modulate2.M, model.modulate3.M]

    phases = []
    for i, M in enumerate(modulators, start=1):
        # 将 M 转换为相位 (0~π)
        phase = (torch.sigmoid(M.detach()) * 2 * np.pi).cpu().numpy()
        phases.append(phase)

        # 保存 txt
        np.savetxt(os.path.join(out_dir, f"phase{i}.txt"), phase, fmt="%.8f")

        # 映射到 0~255 保存 bmp
        img = (phase / (2 * np.pi) * 255).clip(0, 255).astype(np.uint8)
        Image.fromarray(img).save(os.path.join(out_dir, f"tensors{i}.bmp"))

    # 可视化显示
    fig, axes = plt.subplots(1, len(phases), figsize=(12, 5))
    for i, (ax, phase) in enumerate(zip(axes, phases), start=1):
        ax.imshow((phase / (2 * np.pi) * 255).astype(np.uint8), cmap='inferno')
        ax.set_title(f'Modulate_layer_{i}')
        ax.axis('off')
    plt.show()
